# Runtime dependencies for btc-range-model
#
# IMPORTANT — do NOT put an upper bound on `streamlit` (or other packages the
# hosting platform manages). Streamlit Community Cloud force-installs its own
# streamlit version; a cap here makes the dependency resolver fail, which aborts
# the whole install and leaves the app running on a base image that is missing
# app-only packages such as plotly (→ ModuleNotFoundError: plotly).
#
# The one hard pin that matters is scikit-learn: the trained model
# (models/inference_assets_ct.joblib) is pickled with scikit-learn 1.8.0, and
# loading it under a different minor version raises InconsistentVersionWarning
# and can segfault when predicting through the Cython tree ensembles. Keep numpy
# and pandas off the bleeding-edge majors the app was not validated against, but
# with generous ranges so they never collide with the platform's own packages.
# Validated on Python 3.12 (pip + uv): numpy 2.1.3 · pandas 2.2.3 ·
# scikit-learn 1.8.0 · scipy 1.14.1 · pyarrow 25 — model loads with no version
# warning and `streamlit run` starts with no segfault.

# Core scientific stack
numpy>=1.26,<2.3          # block numpy 2.3/2.4 bleeding edge; wide enough to co-resolve
pandas>=2.1,<3            # block pandas 3.x (untested majors); allow all 2.x
scipy>=1.11
scikit-learn==1.8.0       # MUST match the pickled model's version
joblib>=1.3

# Data feeds
yfinance>=0.2.40
requests>=2.31

# Web app  (no upper bound on streamlit — the platform manages its version)
streamlit>=1.30
plotly>=5.18
# Live auto-refresh of the Overall Trading price column (optional — the app
# degrades gracefully to click/reload refresh if this is unavailable).
streamlit-autorefresh>=1.0.1

# Notebook generation
nbformat>=5.9
